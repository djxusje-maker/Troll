        notify("Invisible ON")
    else
        for obj,v in pairs(originalTransparency) do
            if obj and obj.Parent then
                if type(v)=="table" then
                    for k,x in pairs(v) do pcall(function() obj[k]=x end) end
                else pcall(function() obj.Transparency=v end) end
            end
        end
        table.clear(originalTransparency)
        notify("Invisible OFF")
    end
end

local function SkidFling(TargetPlayer)
    local Character=LocalPlayer.Character
    local Humanoid=Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart=Humanoid and Humanoid.RootPart
    local TCharacter=TargetPlayer.Character
    if not TCharacter then return end
    local THumanoid=TCharacter:FindFirstChildOfClass("Humanoid")
    local TRootPart=THumanoid and THumanoid.RootPart
    local THead=TCharacter:FindFirstChild("Head")
    local Accessory=TCharacter:FindFirstChildOfClass("Accessory")
    local Handle=Accessory and Accessory:FindFirstChild("Handle")
    if Character and Humanoid and RootPart then
        if RootPart.Velocity.Magnitude<50 then getgenv().OldPos=RootPart.CFrame end
        if THumanoid and THumanoid.Sit then notify(TargetPlayer.Name.." is sitting");return end
        if TRootPart then workspace.CurrentCamera.CameraSubject=THumanoid
        elseif THead then workspace.CurrentCamera.CameraSubject=THead
        elseif Handle then workspace.CurrentCamera.CameraSubject=Handle end
        if not TCharacter:FindFirstChildWhichIsA("BasePart") then return end

        local function FPos(BasePart,Pos,Ang)
            RootPart.CFrame=CFrame.new(BasePart.Position)*Pos*Ang
            Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position)*Pos*Ang)
            RootPart.Velocity=Vector3.new(9e7,9e7*10,9e7)
            RootPart.RotVelocity=Vector3.new(9e9,9e9,9e9)
        end
        local function SFBasePart(BasePart)
            local Time=tick()
            local Angle=0
            repeat
                if RootPart and THumanoid then
                    Angle=Angle+200
                    FPos(BasePart,CFrame.new(0,1.5,0),CFrame.Angles(math.rad(Angle),0,0));task.wait()
                    FPos(BasePart,CFrame.new(0,-1.5,0),CFrame.Angles(math.rad(Angle),0,0));task.wait()
                end
            until tick()-Time>2 or not flingActive
        end

        workspace.FallenPartsDestroyHeight=0/0
        local BV=Instance.new("BodyVelocity")
        BV.Parent=RootPart
        BV.Velocity=Vector3.new(0,0,0)
        BV.MaxForce=Vector3.new(9e9,9e9,9e9)
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated,false)
        if TRootPart then SFBasePart(TRootPart)
        elseif THead then SFBasePart(THead)
        elseif Handle then SFBasePart(Handle)
        else notify(TargetPlayer.Name.." has no valid parts");return end
        BV:Destroy()
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated,true)
        workspace.CurrentCamera.CameraSubject=Humanoid
        if getgenv().OldPos then
            repeat
                RootPart.CFrame=getgenv().OldPos*CFrame.new(0,.5,0)
                Character:SetPrimaryPartCFrame(getgenv().OldPos*CFrame.new(0,.5,0))
                Humanoid:ChangeState("GettingUp")
                for _,part in pairs(Character:GetChildren()) do
                    if part:IsA("BasePart") then part.Velocity,part.RotVelocity=Vector3.new(),Vector3.new() end
                end
                task.wait()
            until (RootPart.Position-getgenv().OldPos.p).Magnitude<25
            workspace.FallenPartsDestroyHeight=getgenv().FPDH
        end
    else
        notify("Your character is not ready")
    end
end

local function StartFling()
    if flingActive then notify("Fling already active!");return end
    local count=0
    for _ in pairs(SelectedTargets) do count+=1 end
    if count==0 then notify("Select a target first!");return end
    flingActive=true
    notify("🌀 Flinging "..count.." target(s)!")
    flingConnection=task.spawn(function()
        while flingActive do
            local validTargets={}
            for p in pairs(SelectedTargets) do
                if p and p.Parent and p.Character then validTargets[p]=p
                else
                    SelectedTargets[p]=nil
                    local cb=PlayerCheckboxes[p]
                    if cb then cb.Checkmark.Visible=false;cb.Button.BackgroundColor3=Colors.Button end
                end
            end
            for p in pairs(validTargets) do
                if flingActive then SkidFling(p);task.wait(0.1) else break end
            end
            task.wait(0.5)
        end
    end)
end
local function StopFling()
    flingActive=false
    if flingConnection then pcall(task.cancel,flingConnection);flingConnection=nil end
    workspace.FallenPartsDestroyHeight=getgenv().FPDH
    notify("⏹ Fling stopped")
end

end

local function teleportToServer(placeId,jobId)
    if not placeId or not jobId then notify("Invalid server");return end
    if tostring(placeId)==tostring(game.PlaceId) and tostring(jobId)==tostring(game.JobId) then notify("Already in this server");return end
    local ok=pcall(function() TeleportService:TeleportToPlaceInstance(tonumber(placeId),tostring(jobId),LocalPlayer) end)
    if not ok then notify("Teleport failed") end
end
local function getRequestFunction()
    return (syn and syn.request) or (http and http.request) or request
end
local function getPublicServers()
    local req=getRequestFunction()
    if not req then return nil end
    local url="https://games.roblox.com/v1/games/"..tostring(game.PlaceId).."/servers/Public?sortOrder=Asc&limit=100"
    local ok,response=pcall(function() return req({Url=url,Method="GET"}) end)
    if not ok or not response or not response.Body then return nil end
    local decoded,data=pcall(function() return HttpService:JSONDecode(response.Body) end)
    if not decoded or not data or not data.data then return nil end
    local servers={}
    for _,s in ipairs(data.data) do
        if s.id and s.playing and s.id~=game.JobId then table.insert(servers,s) end
    end
    return servers
end
local function joinEmptyServer()
    notify("Finding low-pop server...")
    task.spawn(function()
        local servers=getPublicServers()
        if not servers or #servers==0 then notify("No servers found");return end
        table.sort(servers,function(a,b) return (a.playing or 999)<(b.playing or 999) end)
        teleportToServer(game.PlaceId,servers[1].id)
    end)
end
local function joinRandomServer()
    notify("Finding random server...")
    task.spawn(function()
        local servers=getPublicServers()
        if not servers or #servers==0 then notify("No servers found");return end
        teleportToServer(game.PlaceId,servers[math.random(1,#servers)].id)
    end)
end
local function joinPlayerServer(username)
    username=tostring(username or ""):gsub("^%s+",""):gsub("%s+$","")
    if username=="" then notify("Enter username");return end
    task.spawn(function()
        notify("Looking up @"..username.."...")
        local ok,userId=pcall(function() return Players:GetUserIdFromNameAsync(username) end)
        if not ok or not userId then notify("Username not found");return end
        local req=getRequestFunction()
        if not req then notify("Request API unavailable");return end
        local body=HttpService:JSONEncode({userIds={userId}})
        local rok,response=pcall(function()
            return req({Url="https://presence.roblox.com/v1/presence/users",Method="POST",
                Headers={["Content-Type"]="application/json"},Body=body})
        end)
        if not rok or not response or not response.Body then notify("Presence failed");return end
        local decoded,data=pcall(function() return HttpService:JSONDecode(response.Body) end)
        local presence=decoded and data and data.userPresences and data.userPresences[1]
        if not presence or not presence.placeId or not presence.gameId then notify("Not in a joinable server");return end
        notify("Joining @"..username)
        teleportToServer(presence.placeId,presence.gameId)
    end)
end

local function ApplyHighlightToPlayer(player)
    if not player or player==LocalPlayer or not player.Character then return end
    if highlightObjects[player] then pcall(function() highlightObjects[player]:Destroy() end) end
    local h=Instance.new("Highlight")
    h.FillColor=highlightColor
    h.OutlineColor=Color3.fromRGB(255,255,255)
    h.FillTransparency=.5
    h.OutlineTransparency=0
    h.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop
    h.Parent=player.Character
    highlightObjects[player]=h
end
local function ApplyESPToAll()
    for _,p in ipairs(Players:GetPlayers()) do if p~=LocalPlayer then ApplyHighlightToPlayer(p) end end
end
local function ClearAllESP()
    for _,h in pairs(highlightObjects) do pcall(function() h:Destroy() end) end
    table.clear(highlightObjects)
end
Players.PlayerAdded:Connect(function(p)
    p.CharacterAdded:Connect(function()
        task.wait(.5)
        if highlightESP then ApplyHighlightToPlayer(p) end
    end)
end)
for _,p in ipairs(Players:GetPlayers()) do
    p.CharacterAdded:Connect(function()
        task.wait(.5)
        if highlightESP then ApplyHighlightToPlayer(p) end
    end)
end

local function CreateFOVCircle()
    if fovCircle then pcall(function() fovCircle:Destroy() end);fovCircle=nil end
    if typeof(Drawing)~="table" and typeof(Drawing)~="userdata" then return end
    local ok,c=pcall(function() return Drawing.new("Circle") end)
    if not ok or not c then return end
    fovCircle=c
    fovCircle.Visible=showFOVCircle
    fovCircle.Radius=fovCircleSize
    fovCircle.Thickness=1
    fovCircle.Color=Color3.fromRGB(255,255,255)
    fovCircle.Transparency=.5
end
local function GetNearestPlayerInFOV()
    local nearest,dist=nil,fovCircleSize
    local center=Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2)
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LocalPlayer and p.Character then
            local part=aimPart=="Head" and p.Character:FindFirstChild("Head") or getRoot(p.Character)
            if part then
                local sp=Camera:WorldToScreenPoint(part.Position)
                local d=(Vector2.new(sp.X,sp.Y)-center).Magnitude
                if sp.Z>0 and d<dist then nearest=p;dist=d end
            end
        end
    end
    return nearest
end
local function StartAimAssist()
    aimAssistEnabled=true
    aimAssistConnection=disconnect(aimAssistConnection)
    aimAssistConnection=RunService.RenderStepped:Connect(function()
        if not aimAssistEnabled then return end
        if fovCircle then
            fovCircle.Visible=showFOVCircle
            fovCircle.Position=Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2)
        end
        aimTarget=GetNearestPlayerInFOV()
        local part=aimTarget and aimTarget.Character and (aimPart=="Head" and aimTarget.Character:FindFirstChild("Head") or getRoot(aimTarget.Character))
        if part then
            local cf=CFrame.lookAt(Camera.CFrame.Position,part.Position)
            Camera.CFrame=aimMode=="FullLock" and cf or Camera.CFrame:Lerp(cf,aimSmoothness)
        end
    end)
    notify("🎯 Aim Assist ON ("..aimMode..")")
end
local function StopAimAssist()
    aimAssistEnabled=false
    aimAssistConnection=disconnect(aimAssistConnection)
    aimTarget=nil
    if fovCircle then fovCircle.Visible=false end
    notify("Aim Assist OFF")
end
local function StartKillAura()
    killAuraEnabled=true
    killAuraConnection=disconnect(killAuraConnection)
    killAuraConnection=RunService.Heartbeat:Connect(function()
        if not killAuraEnabled then return end
        local c=LocalPlayer.Character
        local root=getRoot(c)
        if not root then return end
        for _,p in ipairs(Players:GetPlayers()) do
            if p~=LocalPlayer and p.Character then
                local tr=getRoot(p.Character)
                if tr and (root.Position-tr.Position).Magnitude<15 then
                    local tool=c:FindFirstChildOfClass("Tool")
                    if tool and tool:FindFirstChild("Handle") then pcall(function() tool:Activate() end) end
                end
            end
        end
    end)
    notify("⚔ Kill Aura ON")
end
local function StopKillAura()
    killAuraEnabled=false
    killAuraConnection=disconnect(killAuraConnection)
    notify("Kill Aura OFF")
end
local function EnableFullbright()
    fullbright=true
    originalLighting.Brightness=Lighting.Brightness
    originalLighting.Ambient=Lighting.Ambient
    originalLighting.OutdoorAmbient=Lighting.OutdoorAmbient
    originalLighting.FogEnd=Lighting.FogEnd
    Lighting.Brightness=2
    Lighting.Ambient=Color3.fromRGB(200,200,200)
    Lighting.OutdoorAmbient=Color3.fromRGB(200,200,200)
    Lighting.FogEnd=100000
    notify("☀ Fullbright ON")
end
local function DisableFullbright()
    fullbright=false
    for k,v in pairs(originalLighting) do pcall(function() Lighting[k]=v end) end
    notify("Fullbright OFF")
end
local function EnableAntiLag()
    antiLag=true
    pcall(function()
        originalGraphics.Quality=settings().Rendering.QualityLevel
        originalGraphics.GlobalShadows=Lighting.GlobalShadows
        originalGraphics.FogEnd=Lighting.FogEnd
        settings().Rendering.QualityLevel=1
        Lighting.GlobalShadows=false
        Lighting.FogEnd=0
        for _,v in ipairs(workspace:GetDescendants()) do
            if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") then
                originalEffects[v]=v.Enabled
                pcall(function() v.Enabled=false end)
            end
        end
    end)
    notify("⚡ Anti-Lag ON")
end
local function DisableAntiLag()
    antiLag=false
    pcall(function()
        settings().Rendering.QualityLevel=originalGraphics.Quality or 10
        Lighting.GlobalShadows=originalGraphics.GlobalShadows or true
        Lighting.FogEnd=originalGraphics.FogEnd or 100000
    end)
    for v,state in pairs(originalEffects) do
        if v and v.Parent then pcall(function() v.Enabled=state end) end
    end
    table.clear(originalEffects)
    notify("Anti-Lag OFF")
end

Resize.Parent=Main

local resizing,resizeStart,resizeStartSize
Resize.InputBegan:Connect(function(input)
    if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
        resizing=true
        resizeStart=input.Position
        resizeStartSize=Main.AbsoluteSize
        input.Changed:Connect(function()
            if input.UserInputState==Enum.UserInputState.End then
                resizing=false
                savedSettings.Size={X=Main.AbsoluteSize.X,Y=Main.AbsoluteSize.Y}
                savedSettings.Position={X=Main.AbsolutePosition.X,Y=Main.AbsolutePosition.Y}
                saveSettings()
            end
        end)
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if not resizing then return end
    if input.UserInputType==Enum.UserInputType.MouseMovement or input.UserInputType==Enum.UserInputType.Touch then
        local d=input.Position-resizeStart
        local newX=math.clamp(resizeStartSize.X+d.X,300,520)
        local newY=math.clamp(resizeStartSize.Y+d.Y,380,700)
        Main.Size=UDim2.fromOffset(newX,newY)
    end
end)

task.spawn(function()
    while ScreenGui.Parent do
        task.wait(2)
        savedSettings.Size={X=Main.AbsoluteSize.X,Y=Main.AbsoluteSize.Y}
        savedSettings.Position={X=Main.AbsolutePosition.X,Y=Main.AbsolutePosition.Y}
        saveSettings()
    end
end)

UserInputService.InputBegan:Connect(function(input,gp)
    if gp then return end
    if input.KeyCode==Enum.KeyCode.RightControl then
        SetGuiVisible(not Main.Visible)
    end
end)
