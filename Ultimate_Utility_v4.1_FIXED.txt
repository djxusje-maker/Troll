local function hasFileAPI()
    return typeof(readfile)=="function" and typeof(isfile)=="function"
end
local function hasWriteAPI()
    return typeof(writefile)=="function"
end
local function readJSON(file)
    if not hasFileAPI() or not isfile(file) then return nil end
    local ok,data = pcall(function() return HttpService:JSONDecode(readfile(file)) end)
    return ok and data or nil
end
local function writeJSON(file,data)
    if not hasWriteAPI() then return false end
    return pcall(function() writefile(file,HttpService:JSONEncode(data)) end)
end

    local hum=getHumanoid(LocalPlayer.Character)

    ApplyWalkSpeed()
    return true

local function EnableNoclip()
    noclip=true
    noclipConnection=disconnect(noclipConnection)
    noclipConnection=RunService.Stepped:Connect(function()
        if not noclip then return end
        local c=LocalPlayer.Character
        if not c then return end
        for _,p in ipairs(c:GetDescendants()) do
            if p:IsA("BasePart") then
                if originalCanCollide[p]==nil then originalCanCollide[p]=p.CanCollide end
                p.CanCollide=false
            end
        end
    end)
    notify("◇ Noclip ON")
end
local function DisableNoclip()
    noclip=false
    noclipConnection=disconnect(noclipConnection)
    for p,v in pairs(originalCanCollide) do if p and p.Parent then p.CanCollide=v end end
    table.clear(originalCanCollide)
    notify("Noclip OFF")
end

local function destroyFlyObjects()
    if flyVelocity then flyVelocity:Destroy();flyVelocity=nil end
    if flyOrientation then flyOrientation:Destroy();flyOrientation=nil end
    if flyAttachment then flyAttachment:Destroy();flyAttachment=nil end
end
local function DisableFly(silent)
    flying=false
    flyConnection=disconnect(flyConnection)
    destroyFlyObjects()
    local hum=getHumanoid(LocalPlayer.Character)
    if hum and hum.Health>0 then
        hum.PlatformStand=false
        hum.AutoRotate=true
        hum:ChangeState(Enum.HumanoidStateType.GettingUp)
        hum.WalkSpeed=walkSpeed
    end
    if not silent then notify("Fly OFF") end
end
local function EnableFly()
    DisableFly(true)
    local c=LocalPlayer.Character
    local hum=getHumanoid(c)
    local root=getRoot(c)
    if not c or not hum or not root or hum.Health<=0 then notify("Character not ready");return end
    flying=true
    hum.PlatformStand=true
    hum.AutoRotate=false
    flyAttachment=Instance.new("Attachment",root)
    flyVelocity=Instance.new("LinearVelocity")
    flyVelocity.Attachment0=flyAttachment
    flyVelocity.RelativeTo=Enum.ActuatorRelativeTo.World
    flyVelocity.VelocityConstraintMode=Enum.VelocityConstraintMode.Vector
    flyVelocity.ForceLimitsEnabled=false
    flyVelocity.VectorVelocity=Vector3.zero
    flyVelocity.Parent=root
    flyOrientation=Instance.new("AlignOrientation")
    flyOrientation.Attachment0=flyAttachment
    flyOrientation.Mode=Enum.OrientationAlignmentMode.OneAttachment
    flyOrientation.RigidityEnabled=false
    flyOrientation.Responsiveness=35
    flyOrientation.MaxTorque=math.huge
    flyOrientation.Parent=root
    flyConnection=RunService.RenderStepped:Connect(function()
        if not flying then return end
        if LocalPlayer.Character~=c or not root.Parent or hum.Health<=0 then DisableFly(true);return end
        local move=hum.MoveDirection
        local flat=Vector3.new(move.X,0,move.Z)
        local look=Camera.CFrame.LookVector
        local right=Camera.CFrame.RightVector
        local flatLook=Vector3.new(look.X,0,look.Z)
        local flatRight=Vector3.new(right.X,0,right.Z)
        flatLook=flatLook.Magnitude>.001 and flatLook.Unit or Vector3.new(0,0,-1)
        flatRight=flatRight.Magnitude>.001 and flatRight.Unit or Vector3.new(1,0,0)
        local desired=Vector3.zero
        if flat.Magnitude>.03 then
            local m=flat.Unit
            local f=m:Dot(flatLook)
            local r=m:Dot(flatRight)
            desired=look*f+flatRight*r
            if desired.Magnitude>.001 then desired=desired.Unit*math.min(flat.Magnitude,1)*flySpeed end
        end
        flyVelocity.VectorVelocity=desired
        if desired.Magnitude>.05 then flyOrientation.CFrame=CFrame.lookAt(root.Position,root.Position+desired.Unit) end
    end)
    notify("3D Fly ON")
end

-- Improved mobile free camera: joystick controls camera movement; touch swipe controls look.
local freeLookConnection,freeMoveConnection
local freeTouchStart
local function StopFreeCamera()
    freeCamera=false
    freeCameraConnection=disconnect(freeCameraConnection)
    freeLookConnection=disconnect(freeLookConnection)
    freeMoveConnection=disconnect(freeMoveConnection)
    Camera.CameraType=Enum.CameraType.Custom
    local hum=getHumanoid(LocalPlayer.Character)
    if hum then Camera.CameraSubject=hum end
    notify("Free Camera OFF")
end
local function StartFreeCamera()
    if freeCamera then return end
    freeCamera=true
    Camera.CameraType=Enum.CameraType.Scriptable
    local cf=Camera.CFrame
    freeCameraPosition=cf.Position
    local look=cf.LookVector
    freeCameraYaw=math.atan2(-look.X,-look.Z)
    freeCameraPitch=math.asin(math.clamp(look.Y,-1,1))

    freeLookConnection=UserInputService.TouchMoved:Connect(function(input,gp)
        if not freeCamera or gp then return end
        local delta=input.Delta
        freeCameraYaw-=delta.X*0.004
        freeCameraPitch=math.clamp(freeCameraPitch-delta.Y*0.004,-1.45,1.45)
    end)
    freeCameraConnection=RunService.RenderStepped:Connect(function(dt)
        if not freeCamera then return end
        local rot=CFrame.Angles(0,freeCameraYaw,0)*CFrame.Angles(freeCameraPitch,0,0)
        local lookVector=rot.LookVector
        local rightVector=rot.RightVector
        local hum=getHumanoid(LocalPlayer.Character)
        local move=hum and hum.MoveDirection or Vector3.zero
        if move.Magnitude>.03 then
            local f=Vector3.new(lookVector.X,0,lookVector.Z)
            local r=Vector3.new(rightVector.X,0,rightVector.Z)
            if f.Magnitude>.01 then f=f.Unit end
            if r.Magnitude>.01 then r=r.Unit end
            freeCameraPosition+=((f*move.Z)+(r*move.X))*50*dt
        end
        Camera.CFrame=CFrame.lookAt(freeCameraPosition,freeCameraPosition+lookVector,Vector3.yAxis)
    end)
    notify("Free Camera ON")
end

local function SetInvisible(on)
    invisible=on
    local c=LocalPlayer.Character
    if not c then return end
    if on then
        for _,obj in ipairs(c:GetDescendants()) do
            if obj:IsA("BasePart") or obj:IsA("Decal") or obj:IsA("Texture") then
                if originalTransparency[obj]==nil then originalTransparency[obj]=obj.Transparency end
                obj.Transparency=1
            elseif obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") then
                if originalTransparency[obj]==nil then originalTransparency[obj]={Enabled=obj.Enabled} end
                obj.Enabled=false
            end
        end
        local hum=getHumanoid(c)
        if hum then
            originalTransparency[hum]={
                NameDisplayDistance=hum.NameDisplayDistance,
                HealthDisplayDistance=hum.HealthDisplayDistance,
                DisplayDistanceType=hum.DisplayDistanceType
            }
            hum.NameDisplayDistance=0
            hum.HealthDisplayDistance=0
            hum.DisplayDistanceType=Enum.HumanoidDisplayDistanceType.None
        end
        notify("👻 Invisible ON")
    else
        for obj,v in pairs(originalTransparency) do
            if obj and obj.Parent then
                if type(v)=="table" then
                    for k,x in pairs(v) do pcall(function() obj[k]=x end) end
                else pcall(function() obj.Transparency=v end) end
            end
        end
        table.clear(originalTransparency)
        notify("Invisible OFF")
    end
end

local function SkidFling(TargetPlayer)

local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")

local TCharacter = TargetPlayer.Character
if not TCharacter then return end

local Accessory = TCharacter:FindFirstChildOfClass("Accessory")

if Character and Humanoid and RootPart then

getgenv().OldPos = RootPart.CFrame
end

if THumanoid and THumanoid.Sit then

-- Fling position with FASTER rotation
local function FPos(BasePart, Pos, Ang)
RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
RootPart.RotVelocity = Vector3.new(9e9, 9e9, 9e9)
end

-- Main loop with DOUBLE rotation speed
local function SFBasePart(BasePart)
local TimeToWait = 2
local Time = tick()
local Angle = 0
repeat
if RootPart and THumanoid then
Angle = Angle + 200 -- FASTER: was 100

if BasePart.Velocity.Magnitude < 50 then
FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
else
FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed * 2), CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed * 2), CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed * 2), CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(Angle), 0, 0))
task.wait()
FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
task.wait()
end
end
until Time + TimeToWait < tick() or not flingActive
end

local function StartFling()
if flingActive then
notify("Fling already active!")
return
end

local count = 0
for _ in pairs(SelectedTargets) do
count = count + 1
end

if count == 0 then
notify("Select at least one target first!")
return
end

flingActive = true
notify("🌀 Flinging " .. count .. " target(s)!")

flingConnection = task.spawn(function()
while flingActive do
local validTargets = {}
for player in pairs(SelectedTargets) do
if player and player.Parent and player.Character then
validTargets[player] = player
else
SelectedTargets[player] = nil
local checkbox = PlayerCheckboxes[player]
if checkbox then
checkbox.Checkmark.Visible = false
checkbox.Button.BackgroundColor3 = Colors.Button
end
end
end

for player in pairs(validTargets) do
if flingActive then
SkidFling(player)
task.wait(0.1)
else
break
end
end
task.wait(0.5)
end
end)
end

local function StopFling()

    if flingConnection then pcall(task.cancel,flingConnection);flingConnection=nil end
    workspace.FallenPartsDestroyHeight=getgenv().FPDH
    notify("⏹️ Fling stopped")
end

local function DisableAntiFling()
    antiFlingEnabled=false
    antiFlingConnection=disconnect(antiFlingConnection)
end
local function EnableAntiFling()
    DisableAntiFling()
    local root=getRoot(LocalPlayer.Character)
    if not root then return end
    antiFlingEnabled=true
    antiFlingConnection=RunService.RenderStepped:Connect(function()
        if not antiFlingEnabled then return end
        local r=getRoot(LocalPlayer.Character)
        local hum=getHumanoid(LocalPlayer.Character)
        if not r or not hum then return end
        local vel=r.AssemblyLinearVelocity
        -- Do not freeze normal movement. Only cancel unusually large external impulses.
        if vel.Magnitude>180 then
            local move=hum.MoveDirection
            local horizontal=Vector3.new(move.X,0,move.Z)
            local keepHorizontal=horizontal.Magnitude>.05 and horizontal.Unit*math.min(walkSpeed,math.max(8,walkSpeed)) or Vector3.zero
            local keepY=math.clamp(vel.Y,-70,70)
            r.AssemblyLinearVelocity=Vector3.new(keepHorizontal.X,keepY,keepHorizontal.Z)
        end
        if r.AssemblyAngularVelocity.Magnitude>120 then
            r.AssemblyAngularVelocity=Vector3.zero
        end
    end)
    notify("🛡️ Anti-Fling ON")
end

local function teleportToServer(placeId,jobId)
    if not placeId or not jobId then notify("Invalid server");return end
    if tostring(placeId)==tostring(game.PlaceId) and tostring(jobId)==tostring(game.JobId) then notify("Already in this server");return end
    local ok=pcall(function() TeleportService:TeleportToPlaceInstance(tonumber(placeId),tostring(jobId),LocalPlayer) end)
    if not ok then notify("Teleport failed") end
end
local function getRequestFunction()
    return (syn and syn.request) or (http and http.request) or request
end
local function getPublicServers(cursor)
    local req=getRequestFunction()
    if not req then return nil end
    local url="https://games.roblox.com/v1/games/"..tostring(game.PlaceId).."/servers/Public?sortOrder=Asc&limit=100"
    if cursor then url=url.."&cursor="..HttpService:UrlEncode(cursor) end
    local ok,response=pcall(function() return req({Url=url,Method="GET"}) end)
    if not ok or not response or not response.Body then return nil end
    local decoded,data=pcall(function() return HttpService:JSONDecode(response.Body) end)
    if not decoded or not data or not data.data then return nil end
    local servers={}
    for _,s in ipairs(data.data) do
        if s.id and s.playing and s.id~=game.JobId then table.insert(servers,s) end
    end
    return servers,data.nextPageCursor
end
local function joinEmptyServer()
    notify("Finding low-pop server...")
    task.spawn(function()
        local servers=getPublicServers()
        if not servers or #servers==0 then notify("No servers found");return end
        table.sort(servers,function(a,b) return (a.playing or 999)<(b.playing or 999) end)
        teleportToServer(game.PlaceId,servers[1].id)
    end)
end
local function joinRandomServer()
    notify("Finding random server...")
    task.spawn(function()
        local servers=getPublicServers()
        if not servers or #servers==0 then notify("No servers found");return end
        teleportToServer(game.PlaceId,servers[math.random(1,#servers)].id)
    end)
end
local function joinPlayerServer(username)
    username=tostring(username or ""):gsub("^%s+",""):gsub("%s+$","")
    if username=="" then notify("Enter username");return end
    task.spawn(function()
        notify("Looking up @"..username.."...")
        local ok,userId=pcall(function() return Players:GetUserIdFromNameAsync(username) end)
        if not ok or not userId then notify("Username not found");return end
        local req=getRequestFunction()
        if not req then notify("Request API unavailable");return end
        local body=HttpService:JSONEncode({userIds={userId}})
        local rok,response=pcall(function()
            return req({Url="https://presence.roblox.com/v1/presence/users",Method="POST",
                Headers={["Content-Type"]="application/json"},Body=body})
        end)
        if not rok or not response or not response.Body then notify("Presence failed");return end
        local decoded,data=pcall(function() return HttpService:JSONDecode(response.Body) end)
        local presence=decoded and data and data.userPresences and data.userPresences[1]
        if not presence or not presence.placeId or not presence.gameId then notify("Not in a joinable server");return end
        notify("Joining @"..username)
        teleportToServer(presence.placeId,presence.gameId)
    end)
end

local function CreateFOVCircle()
    if fovCircle then pcall(function() fovCircle:Destroy() end);fovCircle=nil end
    if typeof(Drawing)~="table" and typeof(Drawing)~="userdata" then return end
    local ok,c=pcall(function() return Drawing.new("Circle") end)
    if not ok or not c then return end
    fovCircle=c
    fovCircle.Visible=showFOVCircle
    fovCircle.Radius=fovCircleSize
    fovCircle.Thickness=1
    fovCircle.Color=Color3.fromRGB(255,255,255)
    fovCircle.Transparency=.5
end
local function GetNearestPlayerInFOV()
    local nearest,dist=nil,fovCircleSize
    local center=Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2)
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LocalPlayer and p.Character then
            local part=aimPart=="Head" and p.Character:FindFirstChild("Head") or getRoot(p.Character)
            if part then
                local sp=Camera:WorldToScreenPoint(part.Position)
                local d=(Vector2.new(sp.X,sp.Y)-center).Magnitude
                if sp.Z>0 and d<dist then nearest=p;dist=d end
            end
        end
    end
    return nearest
end
local function StartAimAssist()
    aimAssistEnabled=true
    aimAssistConnection=disconnect(aimAssistConnection)
    aimAssistConnection=RunService.RenderStepped:Connect(function()
        if not aimAssistEnabled then return end
        if fovCircle then
            fovCircle.Visible=showFOVCircle
            fovCircle.Position=Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2)
        end
        aimTarget=GetNearestPlayerInFOV()
        local part=aimTarget and aimTarget.Character and (aimPart=="Head" and aimTarget.Character:FindFirstChild("Head") or getRoot(aimTarget.Character))
        if part then
            local cf=CFrame.lookAt(Camera.CFrame.Position,part.Position)
            Camera.CFrame=aimMode=="FullLock" and cf or Camera.CFrame:Lerp(cf,aimSmoothness)
        end
    end)
    notify("🎯 Aim Assist ON ("..aimMode..")")
end
local function StopAimAssist()
    aimAssistEnabled=false
    aimAssistConnection=disconnect(aimAssistConnection)
    aimTarget=nil
    if fovCircle then fovCircle.Visible=false end
    notify("Aim Assist OFF")
end

local function EnableFullbright()
    if not next(originalLighting) then
        originalLighting={Brightness=Lighting.Brightness,Ambient=Lighting.Ambient,OutdoorAmbient=Lighting.OutdoorAmbient,FogEnd=Lighting.FogEnd}
    end
    fullbright=true
    Lighting.Brightness=2
    Lighting.Ambient=Color3.fromRGB(200,200,200)
    Lighting.OutdoorAmbient=Color3.fromRGB(200,200,200)
    if not antiLag then Lighting.FogEnd=100000 end
    notify("☀️ Fullbright ON")
end
local function DisableFullbright()
    fullbright=false
    if not antiLag then
        for k,v in pairs(originalLighting) do pcall(function() Lighting[k]=v end) end
    end
    notify("Fullbright OFF")
end

local function EnableAntiLag()
    antiLag=true
    if not originalGraphics.Quality then
        pcall(function() originalGraphics.Quality=settings().Rendering.QualityLevel end)
        originalGraphics.GlobalShadows=Lighting.GlobalShadows
        originalGraphics.FogEnd=Lighting.FogEnd
    end
    pcall(function()
        settings().Rendering.QualityLevel=1
        Lighting.GlobalShadows=false
        Lighting.FogEnd=0
        table.clear(originalEffects)
        for _,v in ipairs(workspace:GetDescendants()) do
            if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") then
                originalEffects[v]=v.Enabled
                v.Enabled=false
            end
        end
    end)
    notify("⚡ Anti-Lag ON")
end
local function DisableAntiLag()
    antiLag=false
    pcall(function()
        if originalGraphics.Quality then settings().Rendering.QualityLevel=originalGraphics.Quality end
        if originalGraphics.GlobalShadows~=nil then Lighting.GlobalShadows=originalGraphics.GlobalShadows end
        if not fullbright and originalGraphics.FogEnd~=nil then Lighting.FogEnd=originalGraphics.FogEnd end
        for obj,v in pairs(originalEffects) do if obj and obj.Parent then obj.Enabled=v end end
        table.clear(originalEffects)
    end)
    if fullbright then Lighting.FogEnd=100000 end
    notify("Anti-Lag OFF")
end
